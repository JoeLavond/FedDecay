\begin{frame}{Introduction to Federated Learning}
    \begin{definition}[Federated Learning (FL)]
        An iterative procedure allowing \textcolor{blue}{multiple users} to build a shared robust machine learning model \textcolor{red}{without sharing data} (e.g. healthcare). 
    \end{definition}

    Key pieces:
    \begin{itemize}
        \item Data: Multiple users with their data
        \item Training: Users each have a local copy of the model for training
        \item \textbf{Communication}: Send model information rather than data \\
            \quad (privacy) \\
            %\quad (privacy and resource demands) \\
    \end{itemize}

\end{frame}

\begin{frame}{Personalized Federated Learning (pFL)}

\textcolor{red}{Q}: Is a single model appropriate if users have very different data sets? \\

\smallskip
\textcolor{blue}{A}: \textcite{pfl-bench} argues for individualized models tailored to each user's data.

\bigskip
How to personalize?
\begin{itemize}
    \item Main works Ditto \parencite{ditto}, FedBN \parencite{fedbn}, and FedEM \parencite{fedem} require additional computation and memory
    \item Simplest solution is fine-tuning from the single Federated Learning model
\end{itemize}
    
Is it possible to find a single model that, after fine-tuning, performs similarly to the more complex solutions?
\end{frame}


\begin{frame}{User Local Training}

    Within a round of FL, perform $K$ steps of GD on the current global model
    \begin{align}
        x^{j+1}=x^j-\textcolor{red}{\alpha}\nabla f(x^j)
    \end{align}

    Motivated by Taylor Expansion of \textcite{reptile}, we propose step size decay (within round)
    \begin{align}
        & x^{j+1}=x^j-\textcolor{blue}{\alpha\beta^{j-1}}\nabla f(x^j) \label{eq: lr-decay} \\
        & x^K = x^0 - \alpha\sum_{j=1}^K \beta^{j-1}\nabla f(x^{j-1}) \nonumber
    \end{align}

    Equations~\ref{eq: lr-decay} is a trade-off between accuracy ($\beta = 0$) and ability to fine-tune ($\beta = 1$). \\
    
\end{frame}
