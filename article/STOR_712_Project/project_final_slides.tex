\begin{frame}<0>{Introduction to Federated Learning}
    \begin{definition}[Federated Learning (FL)]
        An iterative procedure allowing \textcolor{blue}{multiple users} to build a shared robust machine learning model \textcolor{red}{without sharing data} (e.g. healthcare). 
    \end{definition}

    Key pieces:
    \begin{itemize}
        \item Data: Multiple users with their data
        \item Training: Users each have a local copy of the model for training
        \item \textbf{Communication}: Send model information rather than data \\
            \quad (privacy and resource demands) \\
    \end{itemize}

\end{frame}

\begin{frame}<0>{Personalized Federated Learning (pFL)}

    \textcolor{red}{Q}: Is a single model appropriate if users have very different data sets? \\
    
    \smallskip
    \textcolor{blue}{A}: \textcite{pfl-bench} argues for individualized models tailored to each user's data.
    
    \bigskip
    How to personalize?
    \begin{itemize}
        \item Main works Ditto \parencite{ditto}, FedBN \parencite{fedbn}, and FedEM \parencite{fedem} require additional computation and memory
        \item Simplest solution is fine-tuning from the single Federated Learning model
    \end{itemize}
        
    Is it possible to find a single model that, after fine-tuning, performs similarly to the more complex solutions?
    
\end{frame}


\begin{frame}{User Local Training}

    Within a round of FL, each user performs $K$ updates to the global model
    \begin{align}
        \theta_{k + 1} = \theta_{k} - \textcolor{red}{\alpha} \nabla F(\theta_{k}) 
    \end{align}

    Motivated by Taylor Expansion of \textcite{reptile}, we propose learning rate scheduling (within each training round)
    \begin{align}
        & \phi_{k + 1} = \phi_{k} - \textcolor{blue}{\alpha \beta_k} \nabla F(\phi_{k}) \label{eq: lr-decay} 
        %\\ & \phi_K - \phi_0 \propto \sum_{k = 1}^K \beta_{k - 1} \nabla F(\phi_{k - 1}) 
    \end{align}

\end{frame}


\begin{frame}[allowframebreaks]{Taylor Analysis}

    Expansion of our proposed update where the gradients and Hessians are evaluated using the current global model.
    \begin{align}
        & \mathbb{E} \left[ \phi_K - \phi_0 \right] \appropto - B(K) \textcolor{blue}{\mathbb{E} \left[ \Tilde{g}_j \right]} + \left( \alpha \sum_{j = 1}^K \beta_{(j - 1)} B(j - 1) \right) \textcolor{blue}{\mathbb{E} [ \Tilde{H}_j \Tilde{g}_h ]}
    \end{align}

    \begin{enumerate}
        \item $\mathbb{E} \left[ \Tilde{g}_j \right]$ - minimize joint training objective (initial model success)
        \item $\mathbb{E} [ \Tilde{H}_j \Tilde{g}_h ]$ - maximizes inner product between gradients of different mini-batches (rapid personalization), see Equation~\ref{eq: AvgGradInner}
    \end{enumerate}
    
    \begin{align}
        & \mathbb{E} [ \Tilde{H}_k \Tilde{g}_l ] 
            = \frac{1}{2} \mathbb{E} [ \Tilde{H}_k \Tilde{g}_l + \Tilde{H}_l \Tilde{g}_k ] 
            = \frac{1}{2} \mathbb{E} \left[ \frac{\partial}{\partial \phi_0} \left( \Tilde{g}_k^T \Tilde{g}_l \right) \right] \label{eq: AvgGradInner}
    \end{align}

\begin{frame}[allowframebreaks]{Convergence Results}

    \textbf{Assumptions}:

    \medskip
    For all users $i \in \{1, \ldots, M\}$, the local objective function $F_i$ 
    \begin{enumerate}
        \item is $\mu_i$-strongly convex 
        \item has $L_i$-Lipschitz continuous gradients.
    \end{enumerate}

    \medskip
    Furthermore, their stochastic gradients
    \begin{enumerate}
        \item have bounded variance
        \item have uniformly bounded expected squared norm, $\mathbb{E} \norm{g_k}^2 \leq M$
    \end{enumerate}


\framebreak
    Need learning rate decay over rounds for convergence.
    \begin{align}
        & \eta_t = \alpha_{\lfloor t / K \rfloor} \beta_{(t \bmod K)} 
            \textrm{ where } \alpha_j = \frac{c}{j + d} \textrm{ for some } c, d > 0 
    \end{align}

    \begin{theorem}
        \label{thm: full}
        Under previous assumptions, even with non-iid local datasets, then
        \begin{align*}
            & \mathbb{E} \left[ F ( \theta_g^{(N)} ) \right] - F^* \leq \dfrac{\kappa}{\textcolor{blue}{N} + d - 2} \times \left( C_0 + C_1 \mathbb{E} \norm{\theta_g^{(1)} - \theta_g^*}^2 \right) 
        \end{align*}
        for $N$ communication rounds with $K$ local update steps.
    \end{theorem}
    
\end{frame}
